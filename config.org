(defun partition (lst)
  (let ((n (length lst)))
    (mapcar (lambda (i) (list (nth i lst) (nth (1+ i) lst)))
            (number-sequence 0 (- n 2) 2))))

    (dotimes (i (length pairs))
      (message "Pair %s is: '%s" i (nth i pairs)))))

  "Cycle the visibility of the current element or its parent at point."
  (interactive)

         (type (org-element-type element)))

            (eq type 'inlinetask)
            (eq type 'item)
            (eq type 'plain-list)
            (eq type 'property-drawer)

        ;; If on a cycleable element, operate there

                                         '(headline inlinetask item plain-list property-drawer)

        (goto-char (org-element-property :begin parent))))

             (not (functionp 'json-rpc-connection))  ;; native json-rpc
             (executable-find "emacs-lsp-booster"))
        (progn
          (when-let ((command-from-exec-path (executable-find (car orig-result))))  ;; resolve command from exec-path (in case not found in $PATH)
            (setcar orig-result command-from-exec-path))
          (message "Using emacs-lsp-booster for %s!" orig-result)
          (cons "emacs-lsp-booster" orig-result))
      orig-result)))

(advice-add 'lsp-resolve-final-command :around #'lsp-booster--advice-final-command)
